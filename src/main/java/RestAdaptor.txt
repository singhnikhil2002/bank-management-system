                } else {

                        String dataKey2 = (String) mapentry2.getKey();

                                        collCountMap.get(dataKey), inmsg);
                            } else {
                                dataValue2 = (mapentry2.getValue() != null)
                                        ? mapentry2.getValue().toString().trim() : "";
                                inmsg.setMoreFieldValue(dataKey2, dataValue2,
                                        collCountMap.get(dataKey));
                            }
                        }
                    }

                    collCountMap.put(dataKey, collCountMap.get(dataKey) + 1);
                }
            }
        }

        String appmsg = inmsg.convFieldsToString();
        ArrayList al = new ArrayList();
        al.add(apphdr);
        al.add(appmsg);
        al.add(tranName);


        return al;
    }


    private void processNestedCollection(Map.Entry<Object, Object> mapEntry,
                                         Integer collcount, MessageDef inmsg) {
        JSONArray collData = (JSONArray) mapEntry.getValue();
        HashMap<String, Integer> collCountMap = new HashMap<String, Integer>();
        collCountMap.put((String) mapEntry.getKey(), 0);

        for (Object collObjData : collData) {
            JSONObject collData1 = (JSONObject) collObjData;
            Iterator<Map.Entry<Object, Object>> collFields = collData1.entrySet()
                    .iterator();

            while (collFields.hasNext()) {
                Map.Entry<Object, Object> mapentry2 = (Map.Entry) collFields.next();
                String dataKey2 = (String) mapentry2.getKey();

                if (dataKey2 != null) {
                    String dataValue2 = "";

                    if (mapentry2.getValue() instanceof JSONArray) {
                        processNestedCollection(mapentry2,
                                collCountMap.get(mapEntry.getKey()), inmsg);
                    } else {
                        dataValue2 = ((String) mapentry2.getValue()).trim();
                        FieldDef fldCurr = (FieldDef) inmsg.getMoreFieldDefs(dataKey2)
                                .get(0);
                        int count = 0;

                        try {
                            count = Integer.parseInt(fldCurr.getCollStart()
                                    .substring(fldCurr.getCollStart()
                                            .indexOf('_') +
                                            1));
                            count = count * collcount;
                        } catch (Exception e) {
                        }

                        inmsg.setMoreFieldValue(dataKey2, dataValue2,
                                collCountMap.get(mapEntry.getKey()) + count);
                    }
                }
            }
            collCountMap.put((String) mapEntry.getKey(),
                    collCountMap.get(mapEntry.getKey()) + 1);
        }
    }

                FieldDef fieldDef = (FieldDef) outmsg.getFieldDefs().get(Count);
                if (addCollTag) {
                    if ((fieldDef.getValue() != null) &&
                            !fieldDef.getValue().trim().equals("")) {
                        ValueFound = true;
                    }
                }
                if (!fieldDef.getCollStart().equals("")) {
                    if (addCollTag) {
                        ElemCollResp = ElemCollResp.addElement(fieldDef.getCollStart());
                        jsonCollNode1.put(fieldDef.getName(),
                                fieldDef.getValue());
                        jsonCollResp1.add(jsonCollNode1);
                    } else {
                        ElemCollResp = ElemRespData.addElement(fieldDef.getCollStart());
                        rsDataObj.put(fieldDef.getCollStart(), jsonCollResp1);
                    }
                    temp = fieldDef.getCollStart();
                    addCollTag = true;
                    ValueFound = false;
                    if (!fieldDef.getCollEnd().equals("")) {
                        ValueFound = true;
                    }
                }
                if (addCollTag) {
                    Element tempNode = ElemCollResp.elementByID(fieldDef.getName());
                    if (tempNode != null) {
                    }
                    Element ElemResp2 = ElemCollResp.addElement(fieldDef.getName());
                    ElemResp2.setText(fieldDef.getValue());
                    jsonCollNode1.put(fieldDef.getName(), fieldDef.getValue());
                } else {
                    try {
                        Element tempNode = ElemRespData.element(fieldDef.getName());
                        if (tempNode == null) {
                            Element ElemResp = ElemRespData.addElement(fieldDef.getName());
                            ElemResp.setText(fieldDef.getValue());
                        } else {
                            tempNode.setText(fieldDef.getValue());
                        }
                        if (rsObj.containsKey("Stat")) {
                            jsonStatResp.put(fieldDef.getName(), fieldDef.getValue());
                        } else {
                            rsDataObj.put(fieldDef.getName(), fieldDef.getValue());
                        }
                    } catch (Exception ex) {
                        log.error("fieldDef.getName():" + fieldDef.getName() +
                                ":hostResponse: " +
                                messageDefValMap.get("hostResponse"));
                    }
                }
                if (fieldDef.getCollEnd().equals(temp) ||
                        ((temp != null) &&
                                fieldDef.getCollEnd().contains(temp) &&
                                fieldDef.getCollEnd().contains(","))) {
                    boolean nodeRemoved = false;

                    if (!ValueFound &&
                            ElemCollResp.getParent().equals(ElemRespData)) {
                        ElemRespData.remove(ElemCollResp);
                        jsonCollNode1.clear();
                        addCollTag = false;
                        nodeRemoved = true;
                    } else if (!ValueFound) {
                        Element tempRes = ElemCollResp;
                        if (fieldDef.getCollEnd().contains(",")) {
                            ElemCollResp = ElemCollResp.getParent().getParent();
                        } else {
                            ElemCollResp = ElemCollResp.getParent();
                        }
                        ElemCollResp.remove(tempRes);
                        nodeRemoved = true;
                    }

                    if (!nodeRemoved) {
                        if (fieldDef.getCollEnd().contains(",")) {
                            ElemCollResp = ElemCollResp.getParent().getParent();
                        } else {
                            ElemCollResp = ElemCollResp.getParent();
                        }
                        temp = ElemCollResp.getName();
                    }
                    if (nodeRemoved) {
                        temp = ElemCollResp.getName();
                    }

                    if ((ElemCollResp != null) &&
                            ElemCollResp.equals(ElemRespData)) {
                        addCollTag = false;
                        jsonCollResp1.add(jsonCollNode1.clone());
                    }
                }
                if(fieldDef.getName().equals("Opt") && (TranNo.equalsIgnoreCase("050217")) )  {
                    OptValue = fieldDef.getValue();
                }
            }
            if (jsonCollNode1.size() > 0) {
                jsonCollResp1.add(jsonCollNode1);
            }
        }
        if (jsonResponseAsString) {
            jsonoutput = XML.toJSONObject(documentResp.asXML(), true).toString();
        } else {
            jsonoutput = XML.toJSONObject(documentResp.asXML()).toString();
        }
        log.error(jsonoutput);
        return jsonoutput;
    }

    private MessageDef setTellerParam(MessageDef msgDef, String strTranNum,
                                      LinkedHashMap<String, String> headerValues) throws DocumentException {
        ArrayList fieldDefList = (ArrayList) msgDef.getFieldDefs();

        for (Object object : fieldDefList) {
            FieldDef fielddef = (FieldDef) object;
            String fieldName = fielddef.getName();
            msgDef.setFieldValue(fieldName, fielddef.getValue());

            if (headerValues.containsKey(fieldName) &&
                    !headerValues.get(fieldName).equalsIgnoreCase("")) { //224121: added to set data from request
                msgDef.setFieldValue(fieldName, headerValues.get(fieldName));
            } else {
                if (fieldName.equalsIgnoreCase("WorkstationNum")) {
                    msgDef.setFieldValue(fieldName, termIdNo);
                } else if (fieldName.equalsIgnoreCase("InstNum")) {
                    msgDef.setFieldValue(fieldName, instNo);
                } else if (fieldName.equalsIgnoreCase("BrchNum")) {
                    msgDef.setFieldValue(fieldName, branchNo);
                } else if (fieldName.equalsIgnoreCase("TellerNum")) {
                    msgDef.setFieldValue(fieldName, tellerNo);
                }
            }
        }
        msgDef.setFieldValue("TrnNum", strTranNum);
        return msgDef;
    }

    private Vector doHostAccess(String appmsg, String apphdr) {
        ConnectionPool ppool = null;

        try {
            ppool = (ConnectionPool) (System.getProperties().get("host.pool"));
        } catch (ClassCastException e) {
            ConnectionPool pool = null;

            int initial = Integer.parseInt(System.getProperty("initialHostConn"));
            int increment = Integer.parseInt(System.getProperty(
                    "incrementHostConn"));
            pool = ConnectionPoolImpl.createPool(initial, increment);
            System.getProperties().put("host.pool", pool);

            ppool = (ConnectionPool) (System.getProperties().get("host.pool"));
            Vector v = null;
            Comm mycomm = ppool.getConnection();
            mycomm.setAppData(appmsg);
            mycomm.setHeader(apphdr);
            v = mycomm.getHost();
            ppool.returnConnection(mycomm);
            mycomm = null;
            ppool = null;

            return v;
        }

        Vector v = null;

        try {
            Comm mycomm = ppool.getConnection();

            mycomm.setAppData(appmsg);
            mycomm.setHeader(apphdr);

            v = mycomm.getHost();
            ppool.returnConnection(mycomm);
            mycomm = null;
            ppool = null;
        } catch (Exception e) {
            e.printStackTrace();
        }

        return v;
    }
